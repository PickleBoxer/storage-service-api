# Storage Service API - Authentication System cURL Commands

# ============================================
# 1. REGISTER NEW USER
# ============================================
curl -k -X POST https://storage-service-api.test/api/register -H "Content-Type: application/json" -H "Accept: application/json" -d "{\"name\":\"John Doe\",\"email\":\"john@example.com\",\"password\":\"password123\",\"password_confirmation\":\"password123\"}"

# ============================================
# 2. LOGIN (Save the token from response)
# ============================================
curl -k -X POST https://storage-service-api.test/api/login -H "Content-Type: application/json" -H "Accept: application/json" -d "{\"email\":\"john@example.com\",\"password\":\"password123\"}"

# ============================================
# 3. GET AUTHENTICATED USER (Replace YOUR_TOKEN with actual token)
# ============================================
curl -k -X GET https://storage-service-api.test/api/user -H "Content-Type: application/json" -H "Accept: application/json" -H "Authorization: Bearer YOUR_TOKEN"
